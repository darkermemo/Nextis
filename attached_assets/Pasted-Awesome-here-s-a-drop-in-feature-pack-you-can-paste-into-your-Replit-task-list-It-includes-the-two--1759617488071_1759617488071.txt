1) Context-Aware Notifications (reschedule + seize free time)

New DB

model Notification {
  id        String   @id @default(cuid())
  userId    String
  kind      String   // suggestStart | rescheduleMiss | sitBreak | water | bedtime
  payload   Json
  scheduled DateTime
  sentAt    DateTime?
  createdAt DateTime @default(now())
}

API
	•	POST /api/notify/ack
Body: { notificationId, action: "accept"|"dismiss"|"snooze", now? }
Effect:
	•	accept: apply suggested plan change (e.g., bring study forward 60m)
	•	snooze: requeue in +15m
	•	dismiss: mark sentAt=now with no plan change

Worker jobs
	•	suggestStartIfFree (runs every 15m):
	•	If free ≥45m and there’s a high-priority item within 48h → enqueue suggestStart notification.
	•	rescheduleMissedBlocks (hourly):
	•	If a block ended and item_done not logged → enqueue rescheduleMiss with a proposed new slot (use planner hook below).

Planner hooks

/app/api/notify/ack/route.ts

import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { applySuggestion } from "@/lib/planner";

export async function POST(req: NextRequest) {
  const { notificationId, action, now } = await req.json();
  const n = await prisma.notification.findUnique({ where: { id: notificationId }});
  if (!n) return NextResponse.json({ error: "not_found" }, { status: 404 });

  if (action === "accept") {
    await applySuggestion(n.payload, now);
  } else if (action === "snooze") {
    // requeue +15m
    // enqueueBull("notification", { ...n, scheduled: addMinutes(now, 15) })
  }
  await prisma.notification.update({ where: { id: n.id }, data: { sentAt: new Date(now ?? Date.now()) }});
  return NextResponse.json({ ok: true });
}
